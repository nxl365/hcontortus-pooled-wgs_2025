# Compress, index, and concatenate per-chromosome FreeBayes VCF files for each sample

## Tools
* bcftools (`bcftools concat` concatenate VCF)
* samtools (`tabix` index, `bgzip` compress VCF)

### Workflow:
per-chromosome VCF files → bgzip compression → tabix indexing → bcftools concat → tabix indexing

This step compresses and indexes each per-chromosome VCF file generated by FreeBayes. For each sample, the compressed per-chromosome VCF files are then concatenated into one sample VCF file and indexed.


## SLURM bash script

### Create the bash script:

```bash
016_freebayes_vc_merge_index.sh
```

### Script content

```bash
#!/bin/bash -l
#SBATCH -A naiss2025-22-***
#SBATCH -n 1          # one task
#SBATCH -c 17          # 1 threads per sample, so 17 threads/ cores in total
#SBATCH -t 01:00:00
#SBATCH -p shared   
#SBATCH -J fb_merge2
#SBATCH --mail-type=All,TIME_LIMIT_90
#SBATCH --mail-user=nanxing.liu@slu.se  

module load bcftools/1.20
module load samtools/1.20  # samtool includes: bgzip, tabix    # only use one thread

output_dir='.../analysis/variant_explore/illumina_HC_wgs_XJ4126/016_freebayes_vc_merge_index'

input_base='.../analysis/variant_explore/illumina_HC_wgs_XJ4126/015_freebayes_vc'

mkdir -p "$output_dir"

# Loop over each sample directory within input_base
for sample_dir in "$input_base"/*; do
        (
        SAMPLE_ID=$(basename "$sample_dir")
        echo "Processing sample: $SAMPLE_ID"

        # in sample_dir folder: Compress and index each VCF file
        for vcf in "$sample_dir"/*.vcf; do
                        echo "doing bgzip $SAMPLE_ID: $vcf"
            bgzip -c "$vcf" > "$vcf.gz"      # keep vcf (important), and Compress it to vcf.gz
            tabix -p vcf "$vcf.gz"           # Index the compressed VCF file
        done

        # in output folder: 
        # Merge all compressed per-chromosome VCF files into one
        bcftools concat -O z -o "$output_dir/${SAMPLE_ID}.vcf.gz" "$sample_dir"/*.vcf.gz

        # Index the merged VCF file for downstream analysis
        tabix -p vcf "$output_dir/${SAMPLE_ID}.vcf.gz"

        ) &  # run all samples in parallel
done

wait # Wait for all background processes to finish.


## NOTE:
# Must keep the original VCF file.
# bgzip -c "$vcf" > "$vcf.gz" creates a compressed copy and preserves the original VCF.
# Running bgzip "$vcf" directly replaces the original VCF, which is risky if compression or indexing fails

```

# Decompress VCF for steps afterwards

## Tools
* samtools

## SLURM bash script

### Create the bash script:
```bash
017_freebayes_vc_merged_decompress.sh
```

### Script content
```bash
#!/bin/bash -l
#SBATCH -A naiss2025-22-***
#SBATCH -n 1          # one task
#SBATCH -c 17          # 1 core per sample, 17 in total
#SBATCH -t 01:00:00
#SBATCH -p shared
#SBATCH -J fb2_decompress
#SBATCH --mail-type=All,TIME_LIMIT_90
#SBATCH --mail-user=nanxing.liu@slu.se


module load samtools/1.20

input_dir='.../analysis/variant_explore/illumina_HC_wgs_XJ4126/016_freebayes_vc_merge_index'

for gz in "$input_dir"/*.vcf.gz; do
	(
	sample=$(basename "$gz")
	sample_name=${sample%%.vcf.gz}

	bgzip -d -c $gz > "$input_dir/${sample_name}.vcf"     # decompressed vcf.gz to vcf , # keep original vcf.gz file as well

	) &  # run all samples in parallel
done

wait # Wait for all background processes to finish.

```
